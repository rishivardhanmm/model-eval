
import json
import math
from pathlib import Path

INPUT_FILE = Path(__file__).with_name("readings.json")


def convert_reading(record):
    """Convert one record to a number."""
    raw_value = record.get("value")

    # Fix: zero is a valid reading.
    if raw_value is None:
        print("Rejected missing or null value")
        return None

    try:
        value = float(raw_value)
    except (TypeError, ValueError):
        print(f"Rejected invalid value: {raw_value!r}")
        return None

    # Fix: reject NaN and positive/negative Infinity.
    if not math.isfinite(value):
        print(f"Rejected non-finite value: {raw_value!r}")
        return None

    return value


def main():
    records = json.loads(
        INPUT_FILE.read_text(encoding="utf-8")
    )

    valid = []
    results = []

    for record in records:
        reading_id = record.get("id", "<missing-id>")
        value = convert_reading(record)

        if value is None:
            results.append({
                "id": reading_id,
                "status": "REJECTED"
            })
        else:
            valid.append({
                "id": reading_id,
                "value": value
            })
            results.append({
                "id": reading_id,
                "status": "ACCEPTED"
            })

    print("VALID JSON OUTPUT")
    print(json.dumps(valid, indent=2, allow_nan=False))

    print("\nPER-CASE RESULTS")
    print(json.dumps(results, indent=2, allow_nan=False))


if __name__ == "__main__":
    main()
