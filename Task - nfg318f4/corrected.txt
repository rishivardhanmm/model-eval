from enum import Enum


class TicketStatus(Enum):
    OPEN = "open"
    PENDING = "pending"
    CLOSED = "closed"


tickets = [
    {"id": 101, "title": "Login issue", "status": TicketStatus.OPEN},
    {"id": 102, "title": "Payment delay", "status": TicketStatus.PENDING},
    {"id": 103, "title": "Account locked", "status": TicketStatus.CLOSED},
]


def process_ticket(ticket):
    if ticket["status"] == TicketStatus.OPEN:
        message = "Ticket is open"
    elif ticket["status"] == TicketStatus.PENDING:
        message = "Ticket is pending"
    elif ticket["status"] == TicketStatus.CLOSED:
        message = "Ticket is closed"
    else:
        message = "Unknown ticket status"

    print(
        f"Ticket {ticket['id']}: {ticket['title']} "
        f"- {ticket['status'].value} ({message})"
    )


if __name__ == "__main__":
    for ticket in tickets:
        process_ticket(ticket)
